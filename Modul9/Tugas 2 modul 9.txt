#include <stdio.h>

int main() {
    
    char A[10] = {'S', 'T', 'T', 'M', 'A', 'N', 'D', 'A', 'L', 'A'};
    char C;
    int count = 0;

    printf("Masukkan karakter yang ingin dicari: ");

    scanf(" %c", &C); 
    for (int i = 0; i < 10; i++) {
        if (A[i] == C) {
            count++;
        }
    }
    if (count > 0) {
        printf("ADA, karakter '%c' ditemukan %d kali.\n", C, count);
    } else {
        printf("TIDAK ADA\n");
    }

    return 0;
}
