#include <stdio.h>

int main() {
  
    int A[11];
    int data[] = {12, 15, 7, 10, 5, 2, 17, 25, 9, 20, 35, 28, 14, 11, 999};
    int i = 0, j = 0;

    while (data[i] != 999 && j < 11) 
    {
        A[j] = data[i];
        i++;
        j++;
    }

    
    printf("Isi array A yang berada di indeks genap:\n");
    for (int k = 0; k < 11; k += 2) {   (0, 2, 4, 6, 8, 10)
        printf("%d ", A[k]);
    }

    printf("\n");

    return 0;
}
